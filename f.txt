<file path="src/app/components/alerts-banner/alerts-banner.component.ts">
import { Component, inject } from '@angular/core';
import { DatePipe } from '@angular/common';
import { RouterLink } from '@angular/router';
import { Alert } from '../../alert';
import { AlertService } from '../../services/alert.service';

@Component({
  selector: 'app-alerts-banner',
  standalone: true,
  imports: [RouterLink, DatePipe],
  templateUrl: './alerts-banner.component.html',
})
export class AlertsBannerComponent {
  private service = inject(AlertService);

  readonly limit = 3;
  alerts: Alert[] = [];
  loading = true;
  error = false;
  showAll = false;

  constructor() {
    this.service.getAlerts().subscribe({
      next: (alerts) => {
        this.alerts = alerts;
        this.loading = false;
      },
      error: () => {
        this.error = true;
        this.loading = false;
      },
    });
  }

  get displayed(): Alert[] {
    return this.showAll ? this.alerts : this.alerts.slice(0, this.limit);
  }
}

import { Component } from '@angular/core';
import { FormsModule } from '@angular/forms';
import { Router } from '@angular/router';
import { AuthService } from '../auth.service';
import { AuthHeader } from './auth-header';

@Component({
  selector: 'app-login',
  imports: [FormsModule, AuthHeader],
  templateUrl: './login.html',
})
export class Login {
  credentials = { email: '', password: '' };
  error = '';

  constructor(private auth: AuthService, private router: Router) {}

  submit() {
    this.error = '';
    this.auth.login(this.credentials.email, this.credentials.password).subscribe({
      next: () => this.router.navigateByUrl('/dashboard'),
      error: () => (this.error = 'Identifiants incorrects'),
    });
  }
}

import { inject, Injectable } from '@angular/core';
import { HttpClient } from '@angular/common/http';
import { map, Observable } from 'rxjs';

@Injectable({
  providedIn: 'root',
})
export class ApplicationMethodService {
  private readonly http = inject(HttpClient);

  getAll(): Observable<ApplicationMethod[]> {
    return this.http.get<unknown>('/api/application_methods').pipe(
      map((response) => {
        const data = response as {
          member?: ApplicationMethod[];
          'hydra:member'?: ApplicationMethod[];
        } | ApplicationMethod[];

          return data;
        }

      }),

import { HttpInterceptorFn } from '@angular/common/http';
import { inject } from '@angular/core';
import { AuthService } from './auth.service';

export const authInterceptor: HttpInterceptorFn = (req, next) => {
  const token = inject(AuthService).getToken();
  const isPublic = req.url === '/auth' || req.url === '/api/register';

  if (!token || isPublic) {
    return next(req);
  }

  return next(req.clone({ setHeaders: { Authorization: `Bearer ${token}` } }));
};

import { Component, inject, output } from '@angular/core';
import { FormsModule } from '@angular/forms';
import { Subject, of } from 'rxjs';
import {
  debounceTime,
  distinctUntilChanged,
  switchMap,
  catchError
} from 'rxjs/operators';

import { Application } from '../../application';
import { ApplicationSearchService } from '../../services/application-search.service';

@Component({
  selector: 'app-application-search',
  standalone: true,
  imports: [FormsModule],
  templateUrl: './application-search.component.html'
})
export class ApplicationSearchComponent {
  private applicationService = inject(ApplicationSearchService);
  private searchTerms = new Subject<string>();

  results = output<Application[] | null>();

  searchTerm = '';
  applications: Application[] = [];
  loading = false;
  error = '';

  constructor() {
    this.searchTerms.pipe(
      debounceTime(300),
      distinctUntilChanged(),
      switchMap(term => {
        if (!term) {
          this.loading = false;
          this.error = '';
          return of([]);
        }

        this.loading = true;
        this.error = '';

        return this.applicationService.searchApplications(term).pipe(
          catchError(() => {
            this.error = 'Pas de candidature correspondante';
            return of([]);
          })
        );
      })
    ).subscribe(applications => {
      this.applications = applications;
      this.loading = false;
      this.results.emit(this.searchTerm.trim() ? applications : null);
    });
  }

  onSearch(term: string): void {
    const value = term.trim();

    if (!value) {
      this.applications = [];
      this.error = '';
      this.loading = false;
      this.searchTerms.next('');
      return;
    }

    this.searchTerms.next(value);
  }
}

import {
  Component,
  EventEmitter,
  Output,
  OnInit,
  inject,
} from '@angular/core';
import { CommonModule } from '@angular/common';
import {
  FormBuilder,
  ReactiveFormsModule,
  Validators,
} from '@angular/forms';
import { HttpErrorResponse } from '@angular/common/http';
import { FormModalComponent } from '../form-modal/form-modal.component';
import { StepperComponent } from '../stepper/stepper.component';
import { CompanyService } from '../../services/company.service';
import { ApplicationMethodService } from '../../services/application-method.service';
import {
  ApplicationService,
  CreatedApplication,
  CreateApplicationPayload,
} from '../../application.service';
import { Company, ApplicationMethod } from '../../types/application';

@Component({
  selector: 'app-application-form',
  standalone: true,
  imports: [
    CommonModule,
    ReactiveFormsModule,
    FormModalComponent,
    StepperComponent,
  ],
  templateUrl: './application-form.component.html',
  styleUrl: './application-form.component.css',
})
export class ApplicationFormComponent implements OnInit {
  private readonly fb = inject(FormBuilder);
  private readonly companyService = inject(CompanyService);
  private readonly methodService = inject(ApplicationMethodService);
  private readonly applicationService = inject(ApplicationService);

  @Output() closed = new EventEmitter<void>();
  @Output() created = new EventEmitter<CreatedApplication>();

  companies: Company[] = [];
  applicationMethods: ApplicationMethod[] = [];

  loadingCompanies = false;
  loadingMethods = false;

  currentStep = 1;
  addingOffer = false;
  loading = false;
  errorMessage = '';

  readonly form = this.fb.group({
    label: ['', Validators.required],
    companyId: ['', Validators.required],
    city: [''],
    applyDate: ['', Validators.required],
    applicationMethodId: [''],
    description: [''],
  });

  readonly offerForm = this.fb.group({
    jobTitle: ['', Validators.required],
    contractType: [''],
    description: ['', Validators.required],
    publicationDate: ['', Validators.required],
    expirationDate: [''],
    url: ['', Validators.pattern(/^https?:\/\/.+/i)],
  });

  get modalTitle(): string {
    return this.addingOffer
      ? 'Ajouter une candidature et une offre'
      : 'Ajouter une candidature';
  }

  ngOnInit(): void {
    this.loadingCompanies = true;

    this.companyService.getAll().subscribe({
      next: (companies) => {
        this.companies = companies;
        this.loadingCompanies = false;
      },
      error: () => {
        this.loadingCompanies = false;
        this.errorMessage =
          'Impossible de charger les entreprises. Veuillez réessayer.';
      },
    });

    this.loadingMethods = true;

    this.methodService.getAll().subscribe({
      next: (methods) => {
        this.applicationMethods = methods;
        this.loadingMethods = false;
      },
      error: () => {
        this.loadingMethods = false;
        this.errorMessage =
          'Impossible de charger les méthodes de candidature.';
      },
    });
  }

  onCompanyChange(): void {
    const companyId = Number(this.form.controls.companyId.value);

    const company = this.companies.find(
      (item) => item.id === companyId,
    );

    this.form.controls.city.setValue(company?.city ?? '');
  }

  nextStep(): void {
    if (this.loading) {
      return;
    }

    this.errorMessage = '';

    const fields = ['label', 'companyId', 'applyDate'] as const;

    fields.forEach((field) => {
      this.form.controls[field].markAsTouched();
    });

    if (fields.some((field) => this.form.controls[field].invalid)) {
      return;
    }

    this.addingOffer = true;
    this.currentStep = 2;
  }

  previousStep(): void {
    if (this.loading) {
      return;
    }

    this.currentStep = 1;
    this.errorMessage = '';
  }

  close(): void {
    if (this.loading) {
      return;
    }

    this.closed.emit();
  }

  submit(): void {
    if (this.loading) {
      return;
    }

    this.errorMessage = '';
    this.form.markAllAsTouched();

    if (this.form.invalid) {
      this.currentStep = 1;
      return;
    }

    const value = this.form.getRawValue();
    const companyId = Number(value.companyId);
    const methodId = value.applicationMethodId
      ? Number(value.applicationMethodId)
      : null;

    if (!Number.isInteger(companyId) || companyId <= 0) {
      this.errorMessage = 'Sélectionne une entreprise valide.';
      this.currentStep = 1;
      return;
    }

    if (
      methodId !== null &&
      (!Number.isInteger(methodId) || methodId <= 0)
    ) {
      this.errorMessage = 'Sélectionne une méthode de candidature valide.';
      this.currentStep = 1;
      return;
    }

    const payload: CreateApplicationPayload = {
      label: value.label!.trim(),
      applyDate: value.applyDate!,
      companyId,
      applicationMethodId: methodId,
      description: value.description?.trim() || null,
    };

    if (this.currentStep === 2 && this.addingOffer) {
      this.offerForm.markAllAsTouched();

      if (this.offerForm.invalid) {
        return;
      }

      const offerValue = this.offerForm.getRawValue();
      const publicationDate = offerValue.publicationDate!;
      const expirationDate = offerValue.expirationDate || null;
      const url = offerValue.url?.trim() || null;

      if (
        expirationDate &&
        expirationDate < publicationDate
      ) {
        this.errorMessage =
          'La date d’expiration doit être postérieure ou égale à la date de publication.';
        return;
      }

      payload.offer = {
        jobTitle: offerValue.jobTitle!.trim(),
        contractType: offerValue.contractType?.trim() || null,
        description: offerValue.description!.trim(),
        publicationDate,
        expirationDate,
        url,
      };
    }

    this.loading = true;

    this.applicationService.create(payload).subscribe({
      next: (created) => {
        this.loading = false;
        this.created.emit(created);
        this.closed.emit();
      },
      error: (error: HttpErrorResponse) => {
        this.loading = false;
        this.errorMessage = this.getErrorMessage(error);
      },
    });
  }

  private getErrorMessage(error: HttpErrorResponse): string {
    const body = error.error;

    if (typeof body === 'string' && body.trim()) {
      return body;
    }

    if (body && typeof body === 'object') {
      if (typeof body.error === 'string' && body.error.trim()) {
        return body.error;
      }

      if (typeof body.detail === 'string' && body.detail.trim()) {
        return body.detail;
      }

      if (typeof body['hydra:description'] === 'string') {
        return body['hydra:description'];
      }

      if (typeof body.title === 'string' && error.status >= 500) {
        return 'Une erreur serveur est survenue. Réessaie plus tard.';
      }
    }

    if (error.status === 401 || error.status === 403) {
      return 'Tu dois être connecté pour enregistrer cette candidature.';
    }

    if (error.status === 404) {
      return "Le point d'accès demandé est introuvable.";
    }

    if (error.status === 422) {
      return 'Les données sont invalides. Vérifie la date et les champs obligatoires.';
    }

    return "L'enregistrement a échoué. Réessaie.";
  }
}

import { tap } from 'rxjs';

@Injectable({ providedIn: 'root' })
export class AuthService {
  constructor(private http: HttpClient) {}

    register(data: { firstname: string; lastname: string; email: string; password: string }) {
    return this.http.post('/api/register', {
      firstname: data.firstname,
      lastname: data.lastname,
      email: data.email,
      plainPassword: data.password,
    });
  }

  login(email: string, password: string) {
    return this.http
      .post<{ token: string }>('/auth', { email, password })
      .pipe(tap((res) => localStorage.setItem('token', res.token)));
  }

  logout() {
    localStorage.removeItem('token');
  }

  getToken() {
    return localStorage.getItem('token');
  }

  isLoggedIn() {
    return !!this.getToken();
  }
}

import { inject, Injectable } from '@angular/core';
import { HttpClient } from '@angular/common/http';
import { map, Observable } from 'rxjs';
import { Application } from './application';

export interface CreateOfferPayload {
  jobTitle: string;
  contractType?: string | null;
  description: string;
  url?: string | null;
  publicationDate: string;
  expirationDate?: string | null;
}

export interface CreateApplicationPayload {
  label: string;
  applyDate: string;
  companyId: number;
  applicationMethodId?: number | null;
  description?: string | null;
  offer?: CreateOfferPayload | null;
}

export interface CreatedApplication {
  id: number;
  label: string;
  jobTitle: string;
  company: string;
  status: string;
  applyDate: string;
  offer?: {
    id: number;
    jobTitle: string;
    contractType: string | null;
    publicationDate: string;
    expirationDate: string | null;
  } | null;
}

@Injectable({ providedIn: 'root' })
export class ApplicationService {
  private http = inject(HttpClient);

  getAll(): Observable<Application[]> {
    return this.http
      .get<any>('/api/applications')
      .pipe(map((r) => (Array.isArray(r) ? r : r.member ?? r['hydra:member'] ?? []) as Application[]));
  }

  create(payload: CreateApplicationPayload): Observable<CreatedApplication> {
    return this.http.post<CreatedApplication>(
      '/api/applications',
      payload
    );
  }
}

import { Component, inject } from '@angular/core';
import { DatePipe } from '@angular/common';
import { Router, RouterLink } from '@angular/router';
import { Application } from '../application';
import { ApplicationService, CreatedApplication } from '../application.service';
import { AuthService } from '../auth.service';
import { ApplicationSearchComponent } from '../components/application-search/application-search.component';
import { StatsPanelComponent } from '../components/stats-panel/stats-panel.component';
import { AlertsBannerComponent } from '../components/alerts-banner/alerts-banner.component';
import { ApplicationFormComponent } from '../components/application-form/application-form.component';

@Component({
  selector: 'app-dashboard',
  imports: [
    RouterLink,
    DatePipe,
    ApplicationSearchComponent,
    StatsPanelComponent,
    AlertsBannerComponent,
    ApplicationFormComponent,
  ],
  templateUrl: './dashboard.html',
})
export class Dashboard {
  private service = inject(ApplicationService);
  private auth = inject(AuthService);
  private router = inject(Router);

  applications: Application[] = [];
  searchResults: Application[] | null = null;
  loading = true;
  error = false;
  page = 1;
  pageSize = 5;
  modalOpen = false;

  constructor() {
    this.service.getAll().subscribe({
      next: (list) => {
        this.applications = list;
        this.loading = false;
      },
      error: (e) => {
        if (e.status === 401) {
          this.logout();
        }
        this.error = true;
        this.loading = false;
      },
    });
  }

  get displayed(): Application[] {
    return this.searchResults ?? this.applications;
  }

  get totalPages(): number {
    return Math.ceil(this.displayed.length / this.pageSize);
  }

  get pages(): number[] {
    return Array.from({ length: this.totalPages }, (_, i) => i + 1);
  }

  get pagedApplications(): Application[] {
    const start = (this.page - 1) * this.pageSize;
    return this.displayed.slice(start, start + this.pageSize);
  }

  onResults(results: Application[] | null): void {
    this.searchResults = results;
    this.page = 1;
  }

  logout(): void {
    this.auth.logout();
    this.router.navigate(['/login']);
  }

  details(): void {
    this.router.navigate(['/details/id']);
  openModal(): void {
    this.modalOpen = true;
  }

  closeModal(): void {
    this.modalOpen = false;
  }

  onApplicationCreated(created: CreatedApplication): void {
    const application: Application = {
      id: created.id,
      company: created.company,
      jobTitle: created.jobTitle,
      status: created.status,
      applyDate: created.applyDate,
    };

    this.applications = [
      application,
      ...this.applications.filter((item) => item.id !== created.id),
    ];

    this.searchResults = null;
    this.page = 1;
    this.closeModal();
  }
}
